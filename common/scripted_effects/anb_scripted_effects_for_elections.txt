election_immediate_pre_effect = {
	hidden_effect = {
		pre_select_possible_ruler_focus = yes
		if = {
			limit = { has_government_attribute = has_limited_terms }
			export_to_variable = {
				which = max_election_term
				value = modifier:max_terms
			}
		}
		if = { limit = { mission_completed = N93_lineage_of_marunuma }
			N93_prepare_all_lineages = yes
		}

		if = {
			limit = {
				has_government_attribute = mana_from_election
			}
			export_to_variable = {
				which = election_cycle_length
				value = modifier:election_cycle
			}
			# placeholder. Right, this assumes an election base duration of 4 years. If you have, say base years of 10, this breaks
			# barumand_land_of_equals_reform is one example reform
			change_variable = { election_cycle_length = 4 } #4 years for base election

			set_variable = {
				which = mana_gained_from_election_new
				which = election_cycle_length
			}
			# You get triple for electing a new person
			multiply_variable = { mana_gained_from_election_new = 3 }

			set_variable = {
				which = mana_gained_from_election_re_elect
				which = election_cycle_length
			}
		}
	}
}

add_mana_on_election = {
	if = {
		limit = { has_government_attribute = mana_from_election }
		custom_tooltip = add_mana_based_on_election_length_$election_type$
		hidden_effect = { change_mana = { var_amount = mana_gained_from_election_$election_type$ } }
	}
}

re_elect_ruler = {
	custom_tooltip = remains_ruler
	if = {
		limit = {
			has_government_attribute = has_limited_terms
		}
		custom_tooltip = current_term_tt
	}
	change_adm = 1
	change_dip = 1
	change_mil = 1

	add_karma = -5
	add_mana_on_election = {
		election_type = re_elect
	}
	
	random_list = {
		10 = {
			add_adm_power = 50
		}
		10 = {
			add_dip_power = 50
		}
		10 = {
			add_mil_power = 50
		}
	}

	if = {
		limit = {
			government = republic
			NOT = { republican_tradition = 20 }
			NOT = { has_government_attribute = cannot_become_dictatorship }
		}
		if = {
			limit = {
				is_revolutionary_republic_trigger = yes
			}
			change_government = monarchy
			add_government_reform = revolutionary_empire_reform
		}
		else_if = {
			limit = {
				has_dlc = "Res Publica"
			}
			add_government_reform = presidential_despot_reform
		}
		else = {
			change_government = monarchy
		}
	}

	if = {
		limit = { # This gov reform prevents dictatorships but can spawn rebels
			has_reform = republic_of_free_peoples
			uses_republican_tradition = yes
			NOT = { republican_tradition = 20 }
		}
		capital_scope = {
			spawn_rebels = {
				type = particularist_rebels
				size = 3
			}
		}
	}
	
	if = {	#long-lived races
		limit = {
			OR = {
				has_country_modifier = gnomish_administration
				has_country_modifier = elven_administration
				has_country_modifier = dwarven_administration
			}
		}
		add_scaled_republican_tradition = -3
	}
	else_if = {
		limit = {
			has_country_flag = increased_election_cost
		}
		add_scaled_republican_tradition = -15
		clr_country_flag = increased_election_cost
	}
	else = {
		add_scaled_republican_tradition = -10
	}
	#Samsumbat (maybe other Republic Steward of the Sorrow later)
	if = {
		limit = {
			is_or_was_tag = { tag = F60 }
			uses_meritocracy = yes
		}
		add_meritocracy = -8
	}
	if = { #Lake Federation Authority
		limit = { has_government_mechanic = lakefed_authority_mechanic }
		add_government_power = {
			mechanic_type = lakefed_authority_mechanic
			power_type = lakefed_authority_power
			value = 20
		}
	}

	if = {
		limit = {
			has_government_attribute = reelection_depowers_estates
		}
		add_estate_influence_modifier = {
			estate = all
			influence = -10
			duration = 1460
			desc = REELCTION_DEPOWER_ESTATES
		}
	}
	if = {
		limit = { has_government_attribute = elections_influence_estates }
		add_estate_loyalty = {
			estate = all
			loyalty = 5
		}
	}
	if = { #Ikogshaantus
		limit = { has_reform = ikogshaantus_khantaar_industrial_council }
		add_estate_loyalty = {
			estate = all
			loyalty = -3
		}
	}
	if = {
		limit = { has_government_attribute = reelection_increases_absolutism }
		tooltip = {
			add_absolutism = 10
		}
		custom_tooltip = ADD_ABSOLUTISM_BASED_ON_ELECTION_TERM
		hidden_effect = {
			for_variable_amount = {
				variable = election_term
				effect = "
					add_absolutism = 10
				"
			}
		}
		hidden_effect = {
			change_variable = {
				which = election_term
				value = 1
			}
		}
	}
	if = {
		limit = {
			has_government_attribute = has_limited_terms
		}
		hidden_effect = {
			change_variable = {
				which = election_term
				value = 1
			}
		}
	}
	
	if = {
		limit = { has_government_attribute = elections_increase_factions_influence }
		add_influence_to_random_faction = {
			influence = 20
		}
	}
	#Molten Eternity
	if = {
		limit = {
			religion = molten_eternity
			has_personal_deity = speaking_memory_tradition
		}
		change_variable = {
			which = moltenEternityObligationVar
			value = 100
		}
	}

	if = { # Kimanis, added here since idk if election_term is save to reuse
		limit = {
			is_or_was_tag = { tag = G61 }
			NOT = { mission_completed = G61_worthy_of_gelmonias }
		}
		hidden_effect = {
			change_variable = {
				which = election_term_G61
				value = 1
			}
		}
	}
	if = { #kimanis
		limit = {
			is_or_was_tag = { tag = G61 }
			has_country_modifier = G61_the_demagogues
			NOT = { has_ruler_flag = G61_re-election_buff } 
			ruler_has_bad_personality = yes
		}
		remove_random_bad_ruler_personality = yes
		set_ruler_flag = G61_re-election_buff
	}

	if = {
		limit = {
			has_country_flag = ynn_forced_reelection
		}
		clr_country_flag = ynn_forced_reelection
	}
	if = { #so Tanizu dont get free 6/6/6
		limit = {
			has_reform = namasoguziza_reform
		}
		add_legitimacy = -25
	}
	#Silver republic
	if = {
		limit = {
			has_reform = reformed_silver_council_reform
		}
		if = {
			limit = {
				has_ruler_flag = is_from_ekha
			}
			402 = {
				add_province_modifier = {
					name = silver_touched_province
					duration = 2920
				}
			}
		}
		if = {
			limit = {
				has_ruler_flag = is_from_khasa
			}
			400 = {
				add_province_modifier = {
					name = silver_touched_province
					duration = 2920
				}
			}
		}
		if = {
			limit = {
				has_ruler_flag = is_from_deshak
			}
			382 = {
				add_province_modifier = {
					name = silver_touched_province
					duration = 2920
				}
			}
		}
		if = {
			limit = {
				has_ruler_flag = is_from_sigilan
			}
			5813 = {
				add_province_modifier = {
					name = silver_touched_province
					duration = 2920
				}
			}
		}
	}
	# End of Silver Republic

	if = { #Lake Federation Authority
		limit = { has_government_mechanic = lakefed_authority_mechanic }
		add_government_power = {
			mechanic_type = lakefed_authority_mechanic
			power_type = lakefed_authority_power
			value = 20
		}
	}
	if = {
		limit = { has_reform = sophocracy_republic }
		grant_random_advisor_gov_rank = yes
	}
}

#This scripted effect for for bonuses upon electing a new ruler. It has a lot of
# optional parameters, and it might not be fired for certain election types where
# gaining buffs is not appropriate
elect_new_ruler_bonus_effects = {
	add_mana_on_election = { election_type = new }
	
	if = { 
		limit = { has_reform = sophocracy_republic }
		grant_random_advisor_gov_rank = yes
	}

	[[karma_bonus]
		add_karma = 10
	]

	[[nepotism_remove_tradition]
		add_scaled_republican_tradition = -4
	]

	[[gain_faction_type_influence]
		if = {
			limit = { has_government_attribute = elections_increase_factions_influence }
			add_influence_to_$gain_faction_type_influence$_faction = {
				influence = 20
			}
		}
	]

	[[gain_specific_faction_influence]
		if = {
			limit = { 
				has_faction = $gain_specific_faction_influence$ 
			}
			add_faction_influence = {
				faction = $gain_specific_faction_influence$ 
				influence = 10
			}				
		}
	]
	
	[[gain_estate_type_loyalty]
		if = {
			limit = { has_government_attribute = elections_influence_estates }
			add_estate_loyalty_by_class_$gain_estate_type_loyalty$ = {
				loyalty = 10
			}
		}
	]

	[[gain_all_estate_loyalty]
		if = {
			limit = { has_government_attribute = elections_influence_estates }
			add_estate_loyalty = {
				estate = all
				loyalty = 10
			}
		}
	]

	[[ruler_focus_type]
		if = {
			limit = { has_government_attribute = has_candidate_bonus }
			assign_ruler_focus = {
				type = $ruler_focus_type$
			}
		}
	]

	[[marunuma_lineage_slot]
		if = { limit = { mission_completed = N93_lineage_of_marunuma }
			N93_add_lineage = { slot = $marunuma_lineage_slot$ }
		}
	]

	if = {
		limit = { has_government_attribute = dwarovar_new_minds_effect }
		random_list = {
			20 = {
				trigger = { has_estate = estate_church }
				add_estate_loyalty = { estate = estate_church loyalty = 5 }
			}
			20 = {
				trigger = { has_estate = estate_nobles }
				add_estate_loyalty = { estate = estate_nobles loyalty = 5 }
			}
			20 = {
				trigger = { has_estate = estate_burghers }
				add_estate_loyalty = { estate = estate_burghers loyalty = 5 }
			}
			20 = {
				trigger = { has_estate = estate_mages }
				add_estate_loyalty = { estate = estate_mages loyalty = 5 }
			}
			20 = {
				trigger = { has_estate = estate_artificers }
				add_estate_loyalty = { estate = estate_artificers loyalty = 5 }
			}
			#no adventurers
		}
	}
}

# These maintenance effects must always be fired when electing a new ruler
elect_new_ruler_maintenance = {
	if = {
		limit = {
			has_government_attribute = has_limited_terms
		}
		hidden_effect = {
			set_variable = {
				which = election_term
				value = 1
			}
		}
	}

	if = { # Kimanis
		limit = {
			is_or_was_tag = { tag = G61 }
			NOT = { mission_completed = G61_worthy_of_gelmonias }
		}
		hidden_effect = {
			set_variable = {
				which = election_term_G61
				value = 0
			}
		}
	}
}
