sudomegi_add_holy_order_variable = {
	change_variable = {
		which = sudomegi_total
		value = $amount$
	}
	if = {
		limit = {
			check_variable = {
				which = sudomegi_total
				value = 150
			}
		}
		sudomegi_upgrade_ancestor_modifiers_to_level_three = yes
		if = {
			limit = {
				has_country_modifier = sudomegi_strength_of_faith_5
			}
			remove_country_modifier = sudomegi_strength_of_faith_5
			add_country_modifier = {
				name = sudomegi_strength_of_faith_6
				duration = -1
			}
		}
	}
	else_if = {
		limit = {
			check_variable = {
				which = sudomegi_total
				value = 125
			}
		}
		if = {
			limit = {
				has_country_modifier = sudomegi_strength_of_faith_4
			}
			remove_country_modifier = sudomegi_strength_of_faith_4
			add_country_modifier = {
				name = sudomegi_strength_of_faith_5
				duration = -1
			}
		}
	}
	else_if = {
		limit = {
			check_variable = {
				which = sudomegi_total
				value = 100
			}
		}
		if = {
			limit = {
				has_country_modifier = sudomegi_strength_of_faith_3
			}
			remove_country_modifier = sudomegi_strength_of_faith_3
			add_country_modifier = {
				name = sudomegi_strength_of_faith_4
				duration = -1
			}
		}
	}
	else_if = {
		limit = {
			check_variable = {
				which = sudomegi_total
				value = 75
			}
		}
		sudomegi_upgrade_ancestor_modifiers_to_level_two = yes
		if = {
		limit = {
			has_country_modifier = sudomegi_strength_of_faith_2
		}
		remove_country_modifier = sudomegi_strength_of_faith_2
		add_country_modifier = {
			name = sudomegi_strength_of_faith_3
			duration = -1
		}
	}
	}
	else_if = {
		limit = {
			check_variable = {
				which = sudomegi_total
				value = 50
			}
		}
		if = {
			limit = {
				has_country_modifier = sudomegi_strength_of_faith
			}
			remove_country_modifier = sudomegi_strength_of_faith
			add_country_modifier = {
				name = sudomegi_strength_of_faith_2
				duration = -1
			}
		}
	}
	else_if = {
		limit = {
			check_variable = {
				which = sudomegi_total
				value = 25
			}
		}
		if = {
			limit = {
				NOT = { has_country_modifier = sudomegi_strength_of_faith }
			}
			add_country_modifier = {
				name = sudomegi_strength_of_faith
				duration = -1
			}
		}
	}
}

sudomegi_grant_ancestor_modifier = {
	if = {
        limit = {
            check_variable = {
                which = sudomegi_total
                value = 150
            }
        }
        add_country_modifier = {
            name = sudomegi_$modifier$_3
            duration = -1
        }
    }
    else_if = {
        limit = {
            check_variable = {
                which = sudomegi_total
                value = 75
            }
        }
        add_country_modifier = {
            name = sudomegi_$modifier$_2
            duration = -1
        }
    }
    else = {
        add_country_modifier = {
            name = sudomegi_$modifier$
            duration = -1
        }
    }
	set_country_flag = hall_of_ancestors_$modifier$
}
