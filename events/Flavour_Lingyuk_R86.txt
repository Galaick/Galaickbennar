country_event = { #The Children of the Fox and the Dragon
	id = flavour_lingyuk.14
	title = flavour_lingyuk.14.t
	desc = flavour_lingyuk.14.d
	picture = ADVISOR_eventPicture

	is_triggered_only = yes
	fire_only_once = yes

	option = {
		name = flavour_lingyuk.14.a
		custom_tooltip = R86_the_dakini_tt
		hidden_effect = {
			xianjie_region = {
				limit = {
					NOT = { country_or_non_sovereign_subject_holds = ROOT }
				}
				add_province_modifier = {
					name = R86_the_children_of_the_fox_and_the_dragon
					duration = -1
				}
			}
		}
		if = {
			limit = {
				any_subject_country = {
					NOT = { is_subject_of_type = tributary_state }
					capital_scope = {
						region = xianjie_region
					}
				}
			}
			custom_tooltip = R86_the_dakini_vassal_claim_tt
			hidden_effect = {
				random_subject_country = {
					limit = {
						NOT = { is_subject_of_type = tributary_state }
						capital_scope = {
							region = xianjie_region
						}
					}
					xianjie_region = {
						limit = {
							NOT = { country_or_non_sovereign_subject_holds = ROOT }
							NOT = { is_core = PREV }
						}
						add_permanent_claim = PREV
					}
				}
			}
		}
		else = {
			custom_tooltip = R86_the_dakini_claim_tt
			hidden_effect = {
				xianjie_region = {
					limit = {
						NOT = { country_or_non_sovereign_subject_holds = ROOT }
						NOT = { is_core = ROOT }
					}
					add_permanent_claim = ROOT
				}
			}
		}
	}
}
