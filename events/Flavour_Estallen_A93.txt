country_event = { #develop estallen
	id = flavour_estallen.7
	title = flavour_estallen.7.t
	desc = flavour_estallen.7.d
	picture = HRE_eventpicture

	is_triggered_only = yes

	option = {
		name = flavour_estallen.7.a
		if = {
			limit = { NOT = { has_global_flag = esmari_fashion_happened } }
			if = {
				limit = { ai = no }
				set_country_flag = estallen_spawns_esmari_fashion
				custom_tooltip = estallen_esmari_fashion.tt
			}
			else = {
				random = {
					chance = 50
					set_country_flag = estallen_spawns_esmari_fashion
				}
			}			
		}
		280 = {
			add_center_of_trade_level  = 1
		}
	}
}

				name = A93_learning_the_method
				duration = -1
			}
		}
	}
}
