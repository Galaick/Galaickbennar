# Feuding between local clans
country_event = {
    id = flavour_zyujyut.200
    title = flavour_zyujyut.200.t
	desc = {
	   trigger = { primary_culture = naamjyut }
	   desc = "flavour_zyujyut.200.d"
	}
	desc = {
	   trigger = { NOT = { primary_culture = naamjyut } }
	   desc = "flavour_zyujyut.200.e"
	}
    picture = AFRICAN_KING_MEDIATING_eventPicture
    is_triggered_only = yes
    
    trigger = {
		capital_scope = { superregion = yanshen_superregion }
		NOT = {
			has_country_modifier = naamjyut_clan_feud_countdown
			AND = {
				primary_culture = naamjyut
				is_at_war = yes
			}
		}
		any_owned_province = {
			superregion = yanshen_superregion
			OR = {
				owner = { primary_culture = naamjyut }
				local_autonomy = 30
			}
			OR = {
				culture = naamjyut
				culture = bokai
				AND = {
					owner = { primary_culture = naamjyut }
					culture_group = yan
					OR = {
						area = xiyun_area
						area = gungi_area
						area = bianyuan_area
						area = bianfang_area
						area = jiangszun_area
						area = tianszi_area
						area = aoshan_area
						area = luoyip_area
					}
				}
			}
		}
    }
	
	immediate = {
		hidden_effect = {
			if = {
				limit = { primary_culture = naamjyut }
				add_country_modifier  = {
					name = naamjyut_clan_feud_countdown
					duration = 7300
					hidden = yes
				}
			}
			else = {
				add_country_modifier  = {
					name = naamjyut_clan_feud_countdown
					duration = 18250
					hidden = yes
				}
			}
			random_owned_province = {
				limit = {
					superregion = yanshen_superregion
					OR = {
						owner = { primary_culture = naamjyut }
						local_autonomy = 30
					}
					OR = {
						culture = naamjyut
						culture = bokai
						AND = {
							owner = { primary_culture = naamjyut }
							culture_group = yan
							OR = {
								area = xiyun_area
								area = gungi_area
								area = bianyuan_area
								area = bianfang_area
								area = jiangszun_area
								area = tianszi_area
								area = aoshan_area
								area = luoyip_area
							}
						}
					}
				}
				save_event_target_as = clan_war_target
			}
		}
	}
    
    # As the big guy it is my duty to mediate peace
    option = {
        name = flavour_zyujyut.200.a
		trigger = { primary_culture = naamjyut }
		add_dip_power = -25
		hidden_effect = {
			set_country_flag = clan_war_naamjyut_dip_mediation
			event_target:clan_war_target = {
				random = { 
					chance = 33
					province_event = { 
						id = flavour_zyujyut.201
						days = 7
					}
				}
			}
		}
    }
    
    # As the big guy I shall send troops in to keep the peace
    option = {
        name = flavour_zyujyut.200.b
		trigger = { primary_culture = naamjyut }
		add_mil_power = -25
		hidden_effect = {
			set_country_flag = clan_war_naamjyut_mil_mediation
			event_target:clan_war_target = {
				random = { 
					chance = 33
					province_event = { 
						id = flavour_zyujyut.201
						days = 7
					}
				}
			}
		}
    }
    
    # Try talk some sense to them
    option = {
        name = flavour_zyujyut.200.c
		trigger = { NOT = { primary_culture = naamjyut } }
		add_dip_power = -75
		hidden_effect = {
			set_country_flag = clan_war_foreign_dip_mediation
			event_target:clan_war_target = {
				random = { 
					chance = 50
					province_event = { 
						id = flavour_zyujyut.201
						days = 7
					}
				}
			}
		}
    }
    
    # Enforce the peace by force
    option = {
        name = flavour_zyujyut.200.f
		trigger = { NOT = { primary_culture = naamjyut } }
		add_mil_power = -75
		hidden_effect = {
			set_country_flag = clan_war_foreign_mil_mediation
			event_target:clan_war_target = {
				random_list = {
					50 = {
						province_event = { 
							id = flavour_zyujyut.201
							days = 7
						}
					}
					5 = {
						province_event = { 
							id = flavour_zyujyut.202
							days = 7
						}
					}
				}
			}
		}
    }
    
    # We have nothing to gain here, I don’t care
    option = {
        name = flavour_zyujyut.200.g
		event_target:clan_war_target = {
			add_devastation = 20
			add_province_modifier  = {
				name = local_naamjyut_clan_war
				duration = 730
			}
		}
    }
}

province_event = {
    id = flavour_zyujyut.202
    title = flavour_zyujyut.202.t
    desc = flavour_zyujyut.202.d
	picture = {
		trigger = { has_dlc = "Emperor" }
		picture = FAILED_REVOLUTION_eventPicture
	}
	picture = {
		trigger = { NOT = { has_dlc = "Emperor" } }
		picture = BATTLE_eventPicture
	}
    is_triggered_only = yes
	
    trigger = {
		owner = { has_country_flag = clan_war_foreign_mil_mediation }
	}
    
    # Its treason then!
    option = {
        name = flavour_zyujyut.201.a
		if = {
			limit = {
				owner = { has_estate = estate_nobles }
			}
			owner = {
				add_estate_loyalty = {
					estate = estate_nobles
					loyalty = -20
				}
			}
		}
		else_if = {
			limit = {
				owner = { has_estate = estate_uppercastes }
			}
			owner = {
				add_estate_loyalty = {
					estate = estate_uppercastes
					loyalty = -20
				}
			}
		}
		else_if = {
			limit = {
				owner = { has_estate = estate_ajgriijarul }
			}
			owner = {
				add_estate_loyalty = {
					estate = estate_ajgriijarul
					loyalty = -20
				}
			}
		}
		#Can't be eunnachs, just no
		spawn_rebels = {
			type = noble_rebels
			size = 2
		}
    }
    
    # Fine! Ill leave those childish savages to fight one another!
    option = {
        name = flavour_zyujyut.201.b
		add_devastation = 20
		add_province_modifier  = {
			name = local_naamjyut_clan_war
			duration = 730
		}
		if = {
			limit = {
				OR = {
					current_age = age_of_absolutism
					current_age = age_of_revolutions
				}
			}
			owner = { add_absolutism = -5 }
		}
		owner = { add_legitimacy_equivalent = { amount = -10 republican_tradition = -5 } }
    }
	
	after = {
		owner = { clr_country_flag = clan_war_foreign_mil_mediation }
	}
}
