
namespace = sudomegi

country_event = {
	id = sudomegi.1
	title =  sudomegi.1.t
	desc = sudomegi.1.d
	picture = DIPLOMACY_SIGNING_eventPicture

	is_triggered_only = yes

	option = {
		name = sudomegi.1.a
		if = {
			limit = {
				has_country_flag = sudomegi_tier_three_admin
			}
			add_adm_power = 200
			add_years_of_income = 2
			add_country_modifier = {
				name = sudomegi_calm_mind
				duration = 3650
			}
		}
		else_if = {
			limit = {
				has_country_flag = sudomegi_tier_two_admin
			}
			add_adm_power = 100
			add_years_of_income = 1
		}
		else = {
			add_adm_power = 50
		}
		
		if = {
			limit = {
				has_country_flag = sudomegi_tier_three_dip
			}
			add_dip_power = 200
			custom_tooltip = sudomegi_ae_tooltip_2
			hidden_effect = {
				every_known_country = {
					limit = { NOT = { is_subject_of = ROOT } }
					reverse_add_aggressive_expansion = {
						who = ROOT
						value = -50
						apply_calc = no
					}
				}
			}
			add_country_modifier = {
				name = sudomegi_strong_body
				duration = 3650
			}
		}
		else_if = {
			limit = {
				has_country_flag = sudomegi_tier_two_dip
			}
			add_dip_power = 100
			custom_tooltip = sudomegi_ae_tooltip_1
			hidden_effect = {
				every_known_country = {
					limit = { NOT = { is_subject_of = ROOT } }
					reverse_add_aggressive_expansion = {
						who = ROOT
						value = -25
						apply_calc = no
					}
				}
			}
		}
		else = {
			add_dip_power = 50
		}

		if = {
			limit = {
				has_country_flag = sudomegi_tier_three_mil
			}
			add_mil_power = 200
			add_yearly_manpower = 1
			add_country_modifier = {
				name = sudomegi_iron_spirit
				duration = 3650
			}
		}
		else_if = {
			limit = {
				has_country_flag = sudomegi_tier_two_mil
			}
			add_mil_power = 100
			add_yearly_manpower = 0.5
		}
		else = {
			add_mil_power = 50
		}

		custom_tooltip = sudomegi_holy_order_charges_tt

		hidden_effect = {
			sudomegi_remove_adm_reflection = yes
			sudomegi_remove_dip_reflection = yes
			sudomegi_remove_mil_reflection = yes
		}
	}
}
