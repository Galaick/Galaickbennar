country_event = {
	id = kidsit.4
	title = kidsit.4.t
	desc = kidsit.4.desc
	picture = COUNCIL_TRENT_eventPicture

	is_triggered_only = yes
	trigger = {
		has_reform = table_of_kings_reform
		ruler_can_revive = no

		NOT = { has_ruler_flag = L67_table_of_kings_elected_during_regency }
	}

	immediate = {
		hidden_effect = {
			set_country_flag = L67_table_of_kings_election_in_progress

		hidden_effect = {
			clr_country_flag = L67_table_of_kings_election_in_progress

			1 = {
				add_ruler_modifier = {
					name = L67_king_of_castles

				add_ruler_modifier = {
					name = L67_king_of_the_wild

			limit = { has_government_attribute = has_improved_ruler_election }

	id = kidsit.104

	desc = kidsit.104.desc

	is_triggered_only = yes
	trigger = {
		mission_completed = L67_make_them_forget
		any_owned_province = { NOT = { religion = ROOT } }
	}

	immediate = {
		hidden_effect = {
			random_owned_province = {
				limit = { NOT = { religion = ROOT } }
				save_event_target_as = heathen_holdout_province
			}
		}

	goto = heathen_holdout_province

		name = kidsit.104.a
